import 'dart:convert';
import 'dart:math';
import 'dart:async';
import 'package:flutter/material.dart';
import 'package:http/http.dart' as http;
import 'package:sensors_plus/sensors_plus.dart';

/// Entry point of the application.
/// Initializes and runs the WindApp widget.
void main() {
  runApp(const WindApp());
}

/// The root widget of the app.
/// It defines the theme, dark mode, and initial home screen.
class WindApp extends StatelessWidget {
  const WindApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Wind API',
      theme: ThemeData(
        primarySwatch: Colors.blue,
        brightness: Brightness.light,
      ),
      darkTheme: ThemeData(brightness: Brightness.dark),
      themeMode:
          ThemeMode
              .system, // Automatically switch theme based on system setting
      home: const WindPage(), // Loads the main content screen
    );
  }
}

/// The main screen widget where wind data is fetched and displayed,
/// along with a visual compass indicating wind direction relative to the device orientation.
class WindPage extends StatefulWidget {
  const WindPage({super.key});

  @override
  State<WindPage> createState() => _WindPageState();
}

class _WindPageState extends State<WindPage> {
  final TextEditingController _cityController = TextEditingController(
    text: 'Paris',
  );

  // Variables to store the wind data retrieved from the API.
  double? _windSpeed;
  int? _windDeg;

  // Loading indicator state to prevent duplicate API calls.
  bool _isLoading = false;

  // Orientation angle of the phone in degrees (0–360°).
  double _deviceHeading = 0;

  // Stream subscription to track changes in magnetic field orientation (compass-like behavior).
  StreamSubscription<MagnetometerEvent>? _magnetometerSubscription;

  // Used to throttle sensor updates to avoid UI overload.
  DateTime _lastUpdate = DateTime.now();

  @override
  void initState() {
    super.initState();

    // Start listening to magnetometer events to continuously update device orientation.
    _magnetometerSubscription = magnetometerEvents.listen((event) {
      final now = DateTime.now();

      // Limit updates to every 200 milliseconds to reduce unnecessary UI refreshes.
      if (now.difference(_lastUpdate).inMilliseconds > 200) {
        // Calculate the heading using arctangent of y and x magnetic vectors.
        final angle = atan2(event.y, event.x) * (180 / pi);

        // Normalize angle between 0 and 360 degrees.
        final heading = (angle + 360) % 360;

        setState(() {
          _deviceHeading = heading;
        });

        _lastUpdate = now;
      }
    });
  }

  @override
  void dispose() {
    // Cancel the magnetometer subscription to avoid memory leaks.
    _magnetometerSubscription?.cancel();
    super.dispose();
  }

  /// Fetches wind data for the given city from OpenWeatherMap.
  /// Extracts and stores wind speed and direction in state variables.
  Future<void> fetchWindData(String city) async {
    setState(() {
      _isLoading = true;
    });

    const apiKey =
        'a3baa5aed122cb20874a1a1710ea7586'; // You can replace this with your own API key
    final url = Uri.parse(
      'https://api.openweathermap.org/data/2.5/weather?q=$city&appid=$apiKey&units=metric',
    );

    try {
      final response = await http.get(url);

      // If the request was successful (200 OK), extract wind data
      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        setState(() {
          _windSpeed = data['wind']['speed']?.toDouble();
          _windDeg = data['wind']['deg']?.toInt();
        });
      } else {
        showError('Error: ${response.statusCode}');
      }
    } catch (e) {
      // Catch connection errors or bad formatting
      showError('Connection error');
    } finally {
      setState(() {
        _isLoading = false;
      });
    }
  }

  /// Displays a snackbar with the provided error message.
  void showError(String message) {
    ScaffoldMessenger.of(
      context,
    ).showSnackBar(SnackBar(content: Text(message)));
  }

  /// Converts the wind direction in degrees to a human-readable cardinal direction string.
  String getWindDirection(int? deg) {
    if (deg == null) return '-';

    const directions = [
      'North',
      'North / North-East',
      'North-East',
      'East / North-East',
      'East',
      'East / South-East',
      'South-East',
      'South / South-East',
      'South',
      'South / South-West',
      'South-West',
      'West / South-West',
      'West',
      'West / North-West',
      'North-West',
      'North / North-West',
    ];

    // Each direction covers 22.5 degrees, we use integer division to determine the index.
    return directions[((deg + 11.25) ~/ 22.5) % 16];
  }

  /// Builds the visual layout of the application.
  /// Displays input field, button, loading indicator, and weather results including the compass.
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Wind Data')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            // Input field to enter the name of the city to fetch weather data.
            TextField(
              controller: _cityController,
              decoration: const InputDecoration(
                labelText: 'City',
                border: OutlineInputBorder(),
              ),
              onSubmitted: fetchWindData,
            ),
            const SizedBox(height: 16),

            // Button to trigger data fetch from the weather API.
            ElevatedButton(
              onPressed:
                  _isLoading ? null : () => fetchWindData(_cityController.text),
              child:
                  _isLoading
                      ? const CircularProgressIndicator(color: Colors.white)
                      : const Text('Get Wind Data'),
            ),
            const SizedBox(height: 16),

            // Scrollable area containing wind speed, direction and the compass.
            Expanded(
              child: Scrollbar(
                thumbVisibility: true,
                child: SingleChildScrollView(
                  child: Column(
                    children: [
                      if (_windSpeed != null && _windDeg != null) ...[
                        const SizedBox(height: 16),

                        // Display the wind speed from the API.
                        Text(
                          'Speed: $_windSpeed m/s',
                          style: Theme.of(context).textTheme.titleLarge,
                          textAlign: TextAlign.center,
                        ),

                        // Display the wind direction in both text and degrees.
                        Text(
                          'Direction: ${getWindDirection(_windDeg)} ($_windDeg°)',
                          style: Theme.of(context).textTheme.titleLarge,
                          textAlign: TextAlign.center,
                        ),

                        const SizedBox(height: 32),

                        // Visual representation of wind direction with compass needle.
                        Center(
                          child: SizedBox(
                            height: 220,
                            width: 220,
                            child: CustomPaint(
                              painter: CompassPainter(
                                windDeg: _windDeg!,
                                deviceHeading: _deviceHeading,
                              ),
                            ),
                          ),
                        ),

                        const SizedBox(height: 16),

                        // Show the actual orientation of the device.
                        Text(
                          'Phone orientation: ${_deviceHeading.toStringAsFixed(1)}°',
                          textAlign: TextAlign.center,
                        ),

                        const SizedBox(height: 16),
                      ],
                    ],
                  ),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

/// Custom painter used to render the compass UI.
/// It draws a circular compass and a red needle pointing in the wind's direction relative to device orientation.
class CompassPainter extends CustomPainter {
  final int windDeg;
  final double deviceHeading;

  CompassPainter({required this.windDeg, required this.deviceHeading});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.width / 2;

    // Paint settings for the outer compass circle
    final circlePaint =
        Paint()
          ..color = Colors.grey.shade300
          ..style = PaintingStyle.stroke
          ..strokeWidth = 3;

    // Paint settings for the wind direction needle
    final needlePaint =
        Paint()
          ..color = Colors.red
          ..strokeWidth = 4
          ..strokeCap = StrokeCap.round;

    // Draw the compass circle
    canvas.drawCircle(center, radius, circlePaint);

    // Calculate the angle of the needle by adjusting windDeg to device orientation
    final relativeAngle = ((windDeg - deviceHeading) - 90) * pi / 180;
    final needleLength = radius * 0.9;

    final needleEnd = Offset(
      center.dx + needleLength * cos(relativeAngle),
      center.dy + needleLength * sin(relativeAngle),
    );

    // Draw the needle pointing toward the wind
    canvas.drawLine(center, needleEnd, needlePaint);

    // Function to draw cardinal direction labels (N, S, E, W)
    void drawCardinal(String text, Offset offset) {
      final tp = TextPainter(
        text: TextSpan(
          text: text,
          style: const TextStyle(fontSize: 16, color: Colors.black),
        ),
        textDirection: TextDirection.ltr,
      );
      tp.layout();
      tp.paint(canvas, offset);
    }

    // Draw each cardinal point
    drawCardinal('N', Offset(center.dx - 8, center.dy - radius + 4));
    drawCardinal('S', Offset(center.dx - 8, center.dy + radius - 20));
    drawCardinal('E', Offset(center.dx + radius - 16, center.dy - 8));
    drawCardinal('W', Offset(center.dx - radius + 4, center.dy - 8));
  }

  @override
  bool shouldRepaint(covariant CompassPainter oldDelegate) {
    return oldDelegate.windDeg != windDeg ||
        oldDelegate.deviceHeading != deviceHeading;
  }
}
